import { useEffect, useState } from "react";
import { View, Text, TextInput, TouchableOpacity, StyleSheet, Alert } from "react-native";
import { router } from "expo-router";
import * as SecureStore from "expo-secure-store";
import * as LocalAuthentication from "expo-local-authentication";
import * as Crypto from "expo-crypto";

async function gerarHash(senha: string) {
  return Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, senha);
}

export default function Login() {
  const [email, setEmail] = useState("");
  const [senha, setSenha] = useState("");
  const [bioDisponivel, setBioDisponivel] = useState(false);

  useEffect(() => {
    (async () => {
      const hw = await LocalAuthentication.hasHardwareAsync();
      const enrolled = await LocalAuthentication.isEnrolledAsync();
      const hash = await SecureStore.getItemAsync("passwordHash");
      setBioDisponivel(hw && enrolled && !!hash);
    })();
  }, []);

  async function entrar() {
    const mail = email.trim().toLowerCase();

    if (!mail || !senha) return Alert.alert("Erro", "Preencha e-mail e senha.");
    if (!/^\S+@\S+\.\S+$/.test(mail)) return Alert.alert("Erro", "E-mail inválido.");
    if (senha.length < 6) return Alert.alert("Erro", "A senha deve ter no mínimo 6 caracteres.");

    const emailSalvo = await SecureStore.getItemAsync("savedEmail");
    const hashSalvo = await SecureStore.getItemAsync("passwordHash");
    const hashDigitado = await gerarHash(senha);

    if (!emailSalvo || !hashSalvo) {
      await SecureStore.setItemAsync("savedEmail", mail);
      await SecureStore.setItemAsync("passwordHash", hashDigitado);
      Alert.alert("Conta criada", "Seu acesso foi cadastrado.");
    } else if (mail !== emailSalvo || hashDigitado !== hashSalvo) {
      return Alert.alert("Erro", "E-mail ou senha incorretos.");
    }

    await SecureStore.setItemAsync("token", "token-fake");
    router.replace("/(tabs)");
  }

  async function entrarComBiometria() {
    const res = await LocalAuthentication.authenticateAsync({
      promptMessage: "Entrar no FinançasApp",
      cancelLabel: "Cancelar",
    });
    if (res.success) {
      await SecureStore.setItemAsync("token", "token-fake");
      router.replace("/(tabs)");
    }
  }

  return (
    <View style={s.container}>
      <Text style={s.title}>FinançasApp</Text>
      <TextInput style={s.input} placeholder="E-mail" autoCapitalize="none"
        keyboardType="email-address" value={email} onChangeText={setEmail} />
      <TextInput style={s.input} placeholder="Senha" secureTextEntry
        value={senha} onChangeText={setSenha} />
      <TouchableOpacity style={s.btn} onPress={entrar}>
        <Text style={s.btnText}>Entrar</Text>
      </TouchableOpacity>
      {bioDisponivel && (
        <TouchableOpacity style={[s.btn, s.btnOutline]} onPress={entrarComBiometria}>
          <Text style={[s.btnText, { color: "#2563eb" }]}>Entrar com Face ID</Text>
        </TouchableOpacity>
      )}
    </View>
  );
}

const s = StyleSheet.create({
  container: { flex: 1, justifyContent: "center", padding: 24, backgroundColor: "#fff" },
  title: { fontSize: 32, fontWeight: "bold", marginBottom: 32, textAlign: "center" },
  input: { borderWidth: 1, borderColor: "#ddd", borderRadius: 10, padding: 14, marginBottom: 12, fontSize: 16 },
  btn: { backgroundColor: "#2563eb", padding: 16, borderRadius: 10, alignItems: "center", marginTop: 8 },
  btnOutline: { backgroundColor: "transparent", borderWidth: 1, borderColor: "#2563eb" },
  btnText: { color: "#fff", fontSize: 16, fontWeight: "600" },
});

export const api = axios.create({
  baseURL: 'http://SEU_IP:3000', // ex: http://192.168.0.15:3000
});

api.interceptors.request.use(async (config) => {
  const token = await SecureStore.getItemAsync('accessToken');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});
